cmake_minimum_required(VERSION 3.20)

project("dtc")

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CMAKE_CXX_STANDARD 20)

set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)

	string(REPLACE "/GR" "/GR-" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
	
	string(REPLACE "/MDd" "/MTd" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
	
	string(REPLACE "/MD" "/MT" CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE}")

	add_compile_options(/GF /GS- /sdl-)

	add_link_options(/MANIFEST:NO)

endif()

include_directories("${CMAKE_CURRENT_SOURCE_DIR}/source")

add_compile_definitions(_HAS_EXCEPTIONS=0)

add_compile_definitions(UNICODE _UNICODE)

add_compile_definitions(NOMINMAX)

add_executable("${PROJECT_NAME}" "Source.cpp")

set_target_properties("${PROJECT_NAME}" PROPERTIES OUTPUT_NAME "${PROJECT_NAME}" PDB_NAME "${PROJECT_NAME}.exe" PREFIX "")

# TEST

enable_testing()

set(TARGET_TESTER "tester")

set(TARGET_TESTER_OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/test_generated.c")

set(TARGET_VALIDATOR "validator")

add_executable("${TARGET_TESTER}" "Test.cpp")

add_custom_command(OUTPUT "${TARGET_TESTER_OUTPUT}" COMMAND "${TARGET_TESTER}" ${TARGET_TESTER_OUTPUT} DEPENDS "${TARGET_TESTER}")

add_library("${TARGET_VALIDATOR}" STATIC EXCLUDE_FROM_ALL "${TARGET_TESTER_OUTPUT}")

add_test(NAME "${TARGET_TESTER}" COMMAND "${TARGET_TESTER}" "${TARGET_TESTER_OUTPUT}")

add_test(NAME "${TARGET_VALIDATOR}" COMMAND ${CMAKE_COMMAND} --build . --target "${TARGET_VALIDATOR}")

if(MSVC)

	target_compile_options("${TARGET_VALIDATOR}" PRIVATE /w)

endif()

# DEMO

set(TARGET_DEMO "demo")

set(DEMO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/demo/main.dtl")

set(DEMO_IMMEDIATE "${CMAKE_CURRENT_SOURCE_DIR}/demo/generated/main.dtl.c")

set(DEMO_OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/demo/generated/build.c")

set_property(DIRECTORY PROPERTY ADDITIONAL_CLEAN_FILES "${CMAKE_CURRENT_SOURCE_DIR}/demo/generated")

file(GLOB_RECURSE DEMO_DTL_DEPENDENCIES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/demo/*.dtl")

if(MSVC)

	add_custom_command(OUTPUT "${DEMO_IMMEDIATE}" COMMAND "${CMAKE_C_COMPILER}" /P /Fi"${DEMO_IMMEDIATE}" "${DEMO_SOURCE}" DEPENDS "${DEMO_DTL_DEPENDENCIES}")

else()

	add_custom_command(OUTPUT "${DEMO_IMMEDIATE}" COMMAND "${CMAKE_C_COMPILER}" -E -P -x c "${DEMO_SOURCE}" -o "${DEMO_IMMEDIATE}" DEPENDS "${DEMO_DTL_DEPENDENCIES}")

endif()

set(TARGET_DEMO_BUILD_COMMAND "${PROJECT_NAME}" "-l" "${DEMO_OUTPUT}" "${DEMO_IMMEDIATE}")

add_custom_command(OUTPUT "${DEMO_OUTPUT}" COMMAND ${TARGET_DEMO_BUILD_COMMAND} DEPENDS "${PROJECT_NAME}" "${DEMO_IMMEDIATE}")

add_executable("${TARGET_DEMO}" EXCLUDE_FROM_ALL "${DEMO_OUTPUT}")
